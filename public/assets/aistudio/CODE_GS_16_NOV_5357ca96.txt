function getServerTimeInfo() {
  const now = new Date();
  return {
    todayStr: Utilities.formatDate(now, TZ_WIB, 'yyyy-MM-dd'),
    timeStr: Utilities.formatDate(now, TZ_WIB, 'HH:mm:ss'),
    hhmmStr: Utilities.formatDate(now, TZ_WIB, 'HH:mm'),
    timestampMs: now.getTime()
  };
}

function getOrCreateTugasFolder() {
  const props = PropertiesService.getScriptProperties();
  const savedId = props.getProperty('DRIVE_FOLDER_TUGAS_ID');
  if (savedId) {
    try {
      const f = DriveApp.getFolderById(savedId);
      if (!f.isTrashed()) return f;
    } catch (e) {}
  }
  const existing = DriveApp.getFoldersByName(FOLDER_TUGAS_NAME);
  if (existing.hasNext()) {
    const folder = existing.next();
    props.setProperty('DRIVE_FOLDER_TUGAS_ID', folder.getId());
    return folder;
  }
  const created = DriveApp.createFolder(FOLDER_TUGAS_NAME);
  created.setSharing(DriveApp.Access.PRIVATE, DriveApp.Permission.NONE);
  props.setProperty('DRIVE_FOLDER_TUGAS_ID', created.getId());
  return created;
}

// === KEAMANAN: MANAJEMEN SESI SERVER & RATE LIMITING ===
function generateSessionToken(userObj) {
  const token = Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '');
  const cache = CacheService.getScriptCache();
  cache.put('SESS_' + token, JSON.stringify({
    role: userObj.role,
    nis: userObj.nis,
    username: userObj.username,
    nama: userObj.nama,
    kelas: userObj.kelas,
    jurusan: userObj.jurusan || '-',
    jk: userObj.jk || 'L'
  }), 21600); // Masa aktif sesi 6 jam
  return token;
}

function verifySessionToken(token) {
  if (!token || typeof token !== 'string' || token.length < 32) return null;
  const cache = CacheService.getScriptCache();
  const raw = cache.get('SESS_' + token);
  if (!raw) return null;
  try {
    return JSON.parse(raw);
  } catch (e) {
    return null;
  }
}

function checkRateLimit(actionKey, limitPerMinute) {
  const cache = CacheService.getScriptCache();
  const key = 'RL_' + actionKey;
  const count = parseInt(cache.get(key) || '0', 10);
  if (count >= limitPerMinute) return false;
  cache.put(key, String(count + 1), 60);
  return true;
}

function getHakAksesConfig() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];

  return data.slice(1).map(r => ({

    admin: String(r[2]).toLowerCase() === 'true',

function simpanHakAksesConfig(token, configArray) {
  const sess = verifySessionToken(token);
  if (!sess || sess.role !== 'admin') return { status: false, message: 'Akses ditolak.' };

  pemeriksaanSheetOtomatis();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('HAK_AKSES_ROLE');
  if (!sheet) return { status: false, message: 'Konfigurasi tidak ditemukan.' };

  const lastRow = sheet.getLastRow();
  if (lastRow > 1) sheet.getRange(2, 1, lastRow - 1, 6).clearContent();

  const rows = configArray.map(c => [
    c.id, c.nama, c.admin ? 'true' : 'false', c.guru ? 'true' : 'false', c.karyawan ? 'true' : 'false', c.siswa ? 'true' : 'false'
  ]);
  if (rows.length > 0) sheet.getRange(2, 1, rows.length, 6).setValues(rows);

  return { status: true, message: 'Pengaturan Hak Akses Role berhasil diperbarui!' };
}

// === KALENDER PENDIDIKAN (LURING / DARING / LIBUR & JAM EFEKTIF) ===
function getKalenderStatusByDate(targetDateStr) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('KALENDER_PENDIDIKAN');
  const cleanTarget = String(targetDateStr || '').trim().substring(0, 10);

  if (sheet) {
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      const rTgl = data[i][0] instanceof Date ? Utilities.formatDate(data[i][0], TZ_WIB, 'yyyy-MM-dd') : String(data[i][0] || '').trim().substring(0, 10);
      if (rTgl === cleanTarget) {
        const modeKbm = String(data[i][1] || 'Luring').trim();
        const statusEfektif = String(data[i][2] || 'Efektif').trim();
        const jamMulai = data[i][3] instanceof Date ? Utilities.formatDate(data[i][3], TZ_WIB, 'HH:mm') : String(data[i][3] || '06:30').trim();
        const jamSelesai = data[i][4] instanceof Date ? Utilities.formatDate(data[i][4], TZ_WIB, 'HH:mm') : String(data[i][4] || '16:00').trim();
        return {
          tanggal: cleanTarget,
          modeKbm: modeKbm,
          statusEfektif: statusEfektif,
          jamMulai: jamMulai,
          jamSelesai: jamSelesai,
          keterangan: String(data[i][5] || ''),
          wajibGps: modeKbm.toLowerCase() === 'luring' && statusEfektif.toLowerCase() === 'efektif'
        };
      }
    }
  }

  // Fallback otomatis berdasarkan hari jika tanggal belum diatur di sheet
  const parts = cleanTarget.split('-');
  let isMinggu = false;
  if (parts.length === 3) {
    const d = new Date(parseInt(parts[0], 10), parseInt(parts[1], 10) - 1, parseInt(parts[2], 10));
    isMinggu = d.getDay() === 0;
  }

  if (isMinggu) {
    return {
      tanggal: cleanTarget,
      modeKbm: 'Libur',
      statusEfektif: 'Tidak Efektif',
      jamMulai: '-',
      jamSelesai: '-',
      keterangan: 'Hari Minggu (Tidak Efektif KBM)',
      wajibGps: false
    };
  }

  return {
    tanggal: cleanTarget,
    modeKbm: 'Luring',
    statusEfektif: 'Efektif',
    jamMulai: '06:30',
    jamSelesai: '16:00',
    keterangan: 'KBM Luring Reguler (Wajib Radius GPS Sekolah)',
    wajibGps: true
  };
}

function simpanKalenderPendidikan(token, payload) {
  const sess = verifySessionToken(token);
  if (!sess || (sess.role !== 'admin' && sess.role !== 'guru')) {
    return { status: false, message: 'Akses ditolak.' };
  }

  pemeriksaanSheetOtomatis();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('KALENDER_PENDIDIKAN');
  const data = sheet.getDataRange().getValues();
  const targetTgl = String(payload.tanggal || '').trim().substring(0, 10);

  if (!targetTgl) return { status: false, message: 'Tanggal wajib diisi.' };

  let foundRow = -1;
  for (let i = 1; i < data.length; i++) {
    const rTgl = data[i][0] instanceof Date ? Utilities.formatDate(data[i][0], TZ_WIB, 'yyyy-MM-dd') : String(data[i][0] || '').trim().substring(0, 10);
    if (rTgl === targetTgl) {
      foundRow = i + 1;
      break;
    }
  }

  const rowValues = [
    targetTgl,
    payload.modeKbm || 'Luring',
    payload.statusEfektif || 'Efektif',
    payload.jamMulai || '06:30',
    payload.jamSelesai || '16:00',
    payload.keterangan || '-'
  ];

  if (foundRow !== -1) {
    sheet.getRange(foundRow, 1, 1, 6).setValues([rowValues]);
    return { status: true, message: `Kalender Pendidikan tanggal ${targetTgl} berhasil diperbarui!` };
  } else {
    sheet.appendRow(rowValues);
    return { status: true, message: `Jadwal Kalender Pendidikan tanggal ${targetTgl} berhasil ditambahkan!` };
  }
}

function hapusKalenderPendidikan(token, rowIdx) {
  const sess = verifySessionToken(token);
  if (!sess || (sess.role !== 'admin' && sess.role !== 'guru')) {
    return { status: false, message: 'Akses ditolak.' };
  }
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('KALENDER_PENDIDIKAN');
  if (sheet && rowIdx > 1 && rowIdx <= sheet.getLastRow()) {
    sheet.deleteRow(rowIdx);
    return { status: true, message: 'Entri Kalender Pendidikan berhasil dihapus.' };
  }
  return { status: false, message: 'Data tidak valid.' };
}

// === AUTENTIKASI AMAN ===
function login(usernameOrNis, password) {
  try {
    const inputVal = String(usernameOrNis || '').trim();
    const inputPass = String(password || '').trim();

    if (!inputVal) return { status: false, message: 'Silakan masukkan NIS atau Username.' };
    if (!checkRateLimit('LOGIN_' + inputVal.toLowerCase(), 15)) {
      return { status: false, message: 'Terlalu banyak percobaan masuk. Coba lagi 1 menit.' };
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // 1. Cek Login Siswa berdasarkan NIS di ABSENSI_DATA_SISWA
    const siswaSheet = ss.getSheetByName('ABSENSI_DATA_SISWA');
    if (siswaSheet) {
      const dataSiswa = siswaSheet.getDataRange().getValues();
      for (let i = 1; i < dataSiswa.length; i++) {
        const nis = String(dataSiswa[i][0] || '').trim();
        if (nis && nis.toLowerCase() === inputVal.toLowerCase()) {
          const userObj = {
            role: 'siswa',
            nama: String(dataSiswa[i][1] || ''),
            nis: nis,
            username: nis,
            kelas: String(dataSiswa[i][2] || ''),
            jurusan: String(dataSiswa[i][3] || ''),
            jk: String(dataSiswa[i][4] || '-'),
            tempatPkl: String(dataSiswa[i][5] || '-'),
            nilaiSm1: String(dataSiswa[i][6] || '-'),
            nilaiSm2: String(dataSiswa[i][7] || '-'),
            statusSm1: String(dataSiswa[i][8] || '-'),
            statusSm2: String(dataSiswa[i][9] || '-'),
            tugasBelumSm1: String(dataSiswa[i][10] || '-'),
            tugasBelumSm2: String(dataSiswa[i][11] || '-')
          };
          const token = generateSessionToken(userObj);
          return { status: true, token: token, user: userObj };
        }
      }
    }

    // 2. Cek Login Pegawai/Admin di ABSENSI_DATA_USER
    const userSheet = ss.getSheetByName('ABSENSI_DATA_USER');
    if (userSheet) {
      const userData = userSheet.getDataRange().getValues();
      for (let i = 1; i < userData.length; i++) {
        const u = String(userData[i][0] || '').trim().toLowerCase();
        const p = String(userData[i][1] || '').trim();
        if (u && u === inputVal.toLowerCase() && p === inputPass) {
          const userObj = {
            role: String(userData[i][2] || 'guru').toLowerCase(),
            nama: String(userData[i][3] || u),
            nis: String(userData[i][4] || u),
            username: u,
            kelas: String(userData[i][5] || 'ALL'),
            jk: String(userData[i][6] || 'L')
          };
          const token = generateSessionToken(userObj);
          return { status: true, token: token, user: userObj };
        }
      }
    }

    return { status: false, message: 'Identitas atau kredensial tidak valid.' };
  } catch (err) {
    return { status: false, message: 'Terjadi gangguan autentikasi server.' };
  }
}

        authenticated: false,

        serverTime: serverTime,
        kalenderHariIni: kalenderHariIni,
        mapel: mapelList
      };
    }

    const hakAkses = getHakAksesConfig();
    const rawKalender = getValues('KALENDER_PENDIDIKAN');
    const kalenderList = rawKalender.slice(1).map((r, idx) => ({

      tanggal: r[0] instanceof Date ? Utilities.formatDate(r[0], TZ_WIB, 'yyyy-MM-dd') : String(r[0] || '').trim().substring(0, 10),
      modeKbm: String(r[1] || 'Luring'),

// 1. Pembatasan Pencarian Identitas Siswa (Maks 10 hasil, min 3 karakter, tanpa data sensitif)
function cariIdentitasSiswa(token, query) {
  const sess = verifySessionToken(token);
  if (!sess) return { status: false, message: 'Sesi tidak valid.' };

  if (!checkRateLimit('SEARCH_' + (sess.nis || sess.username), 25)) {
    return { status: false, message: 'Terlalu banyak permintaan pencarian.' };
  }

  const q = String(query || '').trim().toLowerCase();
  if (q.length < 3) {
    return { status: false, message: 'Masukkan minimal 3 karakter NIS atau Nama.' };
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('ABSENSI_DATA_SISWA');
  if (!sheet) return { status: true, results: [] };

  const data = sheet.getDataRange().getValues();
  const results = [];

  for (let i = 1; i < data.length; i++) {
    const nis = String(data[i][0] || '').trim();
    const nama = String(data[i][1] || '').trim();
    const kelas = String(data[i][2] || '').trim();
    const jurusan = String(data[i][3] || '').trim();

    if (!nis) continue;

    // Jika role siswa, hanya izinkan mencari identitas dirinya sendiri
    if (sess.role === 'siswa' && nis.toLowerCase() !== String(sess.nis).trim().toLowerCase()) {
      continue;
    }

    if (nis.toLowerCase().includes(q) || nama.toLowerCase().includes(q)) {
      results.push({ nis: nis, nama: nama, kelas: kelas, jurusan: jurusan });
      if (results.length >= 10) break; // Pembatasan maksimal 10 hasil
    }
  }

  return { status: true, results: results };
}

// 2. Sanitasi Nama File
function sanitizeFileName(rawName) {
  const base = String(rawName || 'dokumen_tugas')
    .replace(/\.[^/.]+$/, '') // hapus ekstensi lama
    .replace(/[^a-zA-Z0-9_-]/g, '_') // hanya alfanumerik, underscore, hyphen
    .replace(/_+/g, '_')
    .replace(/^_|_$/g, '');
  return (base || 'tugas').substring(0, 50);
}

// 3. Validasi Magic Bytes File di Server
function validateMagicBytes(bytes, mimeType) {
  if (!bytes || bytes.length < 8) return false;
  const u = bytes.slice(0, 12).map(b => (b < 0 ? b + 256 : b));

  if (mimeType === 'application/pdf') {
    // %PDF (0x25 0x50 0x44 0x46)
    return u[0] === 0x25 && u[1] === 0x50 && u[2] === 0x44 && u[3] === 0x46;
  }
  if (mimeType === 'application/vnd.openxmlformats-officedocument.wordprocessingml.document') {
    // ZIP/DOCX PK\x03\x04
    return u[0] === 0x50 && u[1] === 0x4B && u[2] === 0x03 && u[3] === 0x04;
  }
  if (mimeType === 'application/msword') {
    // OLE2 DOC D0 CF 11 E0
    return u[0] === 0xD0 && u[1] === 0xCF && u[2] === 0x11 && u[3] === 0xE0;
  }
  if (mimeType === 'image/jpeg') {
    return u[0] === 0xFF && u[1] === 0xD8 && u[2] === 0xFF;
  }
  if (mimeType === 'image/png') {
    return u[0] === 0x89 && u[1] === 0x50 && u[2] === 0x4E && u[3] === 0x47;
  }
  if (mimeType === 'image/webp') {
    // RIFF....WEBP
    return u[0] === 0x52 && u[1] === 0x49 && u[2] === 0x46 && u[3] === 0x46 &&
           u[8] === 0x57 && u[9] === 0x45 && u[10] === 0x42 && u[11] === 0x50;
  }
  return false;
}

// 4. Endpoint Pengiriman Tugas Digital Siswa
function submitTugasDigital(token, payload) {
  const lock = LockService.getScriptLock();
  try {
    const sess = verifySessionToken(token);
    if (!sess || sess.role !== 'siswa') {
      return { status: false, message: 'Akses ditolak. Silakan masuk ulang sebagai Siswa.' };
    }

    if (!checkRateLimit('SUBMIT_TUGAS_' + sess.nis, 5)) {
      return { status: false, message: 'Batas pengiriman tercapai. Tunggu 1 menit.' };
    }

    // Pencegahan Double Submit via Nonce
    const nonce = String(payload.nonce || '').trim();
    if (!nonce || nonce.length < 10) {
      return { status: false, message: 'Token transaksi tidak valid.' };
    }
    const cache = CacheService.getScriptCache();
    if (cache.get('NONCE_' + nonce)) {
      return { status: false, message: 'Pengiriman duplikat terdeteksi. Tugas Anda sudah diproses.' };
    }

    lock.waitLock(15000);
    if (cache.get('NONCE_' + nonce)) {
      lock.releaseLock();
      return { status: false, message: 'Pengiriman duplikat terdeteksi.' };
    }

    // Validasi NIS di Server terhadap ABSENSI_DATA_SISWA & Sesi Aktif
    const inputNis = String(payload.nis || '').trim();
    if (!inputNis || inputNis.toLowerCase() !== String(sess.nis).trim().toLowerCase()) {
      lock.releaseLock();
      return { status: false, message: 'Validasi identitas siswa gagal.' };
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const siswaSheet = ss.getSheetByName('ABSENSI_DATA_SISWA');
    const dataSiswa = siswaSheet ? siswaSheet.getDataRange().getValues() : [];
    let verifiedSiswa = null;

    for (let i = 1; i < dataSiswa.length; i++) {
      const rNis = String(dataSiswa[i][0] || '').trim();
      if (rNis.toLowerCase() === inputNis.toLowerCase()) {
        verifiedSiswa = {
          nis: rNis,
          nama: String(dataSiswa[i][1] || '').trim(),
          kelas: String(dataSiswa[i][2] || '').trim()
        };
        break;
      }
    }

    if (!verifiedSiswa) {
      lock.releaseLock();
      return { status: false, message: 'Identitas siswa tidak terdaftar di database.' };
    }

    // Validasi Mata Pelajaran terhadap MASTER_MAPEL
    const mapelInput = String(payload.mapel || '').trim();
    const mapelSheet = ss.getSheetByName('MASTER_MAPEL');
    const validMapel = mapelSheet
      ? mapelSheet.getDataRange().getValues().slice(1).map(r => String(r[0] || '').trim())
      : [];
    if (!mapelInput || !validMapel.includes(mapelInput)) {
      lock.releaseLock();
      return { status: false, message: 'Mata pelajaran tidak valid.' };
    }

    // Validasi Detail Tugas
    const tanggalTugas = String(payload.tanggalTugas || '').trim().substring(0, 10);
    const keterangan = String(payload.keterangan || '').trim().substring(0, 500);
    if (!/^\d{4}-\d{2}-\d{2}$/.test(tanggalTugas) || keterangan.length < 3) {
      lock.releaseLock();
      return { status: false, message: 'Tanggal dan detail tugas wajib diisi dengan benar.' };
    }

    // Validasi Tipe & Ukuran File di Server
    const mimeType = String(payload.mimeType || '').trim().toLowerCase();
    const ext = ALLOWED_MIME_TYPES[mimeType];
    if (!ext) {
      lock.releaseLock();
      return { status: false, message: 'Format file tidak diizinkan. Gunakan PDF, DOC, DOCX, JPG, JPEG, PNG, atau WEBP.' };
    }

    const base64Data = String(payload.fileBase64 || '');
    if (!base64Data) {
      lock.releaseLock();
      return { status: false, message: 'Berkas dokumen tidak ditemukan.' };
    }

    const decodedBytes = Utilities.base64Decode(base64Data);
    if (!decodedBytes || decodedBytes.length === 0 || decodedBytes.length > MAX_FILE_SIZE_BYTES) {
      lock.releaseLock();
      return { status: false, message: 'Ukuran file melebihi batas maksimal 10 MB atau berkas kosong.' };
    }

    if (!validateMagicBytes(decodedBytes, mimeType)) {
      lock.releaseLock();
      return { status: false, message: 'Isi berkas tidak sesuai dengan format dokumen yang diizinkan.' };
    }

    // Sanitasi Nama File & Simpan ke Google Drive
    const now = new Date();
    const tglCompact = Utilities.formatDate(now, TZ_WIB, 'yyyyMMdd');
    const waktuKirim = Utilities.formatDate(now, TZ_WIB, 'yyyy-MM-dd HH:mm:ss');
    const cleanOrig = sanitizeFileName(payload.fileName);
    const cleanMapel = sanitizeFileName(mapelInput).substring(0, 20);
    const finalFileName = `TGS_${verifiedSiswa.nis}_${cleanMapel}_${tglCompact}_${cleanOrig}.${ext}`;

    const blob = Utilities.newBlob(decodedBytes, mimeType, finalFileName);
    const folder = getOrCreateTugasFolder();
    const savedFile = folder.createFile(blob);
    savedFile.setSharing(DriveApp.Access.PRIVATE, DriveApp.Permission.NONE);

    // Buat Nomor ID Unik Pengumpulan & Catat ke Sheet PENGUMPULAN_TUGAS
    const randomCode = Utilities.getUuid().replace(/-/g, '').substring(0, 6).toUpperCase();
    const idPengumpulan = `TGS-${tglCompact}-${randomCode}`;

    pemeriksaanSheetOtomatis();
    const tugasSheet = ss.getSheetByName('PENGUMPULAN_TUGAS');
    tugasSheet.appendRow([
      idPengumpulan,
      waktuKirim,
      verifiedSiswa.nis,
      verifiedSiswa.nama,
      verifiedSiswa.kelas,
      tanggalTugas,
      mapelInput,
      keterangan,
      finalFileName,
      savedFile.getId(),
      'Terkirim'
    ]);

    // Tandai nonce terpakai selama 10 menit
    cache.put('NONCE_' + nonce, '1', 600);
    lock.releaseLock();

    // Kembalikan konfirmasi tanpa mengekspos File ID atau URL Drive internal
    return {
      status: true,
      receipt: {
        idPengumpulan: idPengumpulan,
        namaSiswa: verifiedSiswa.nama,
        nis: verifiedSiswa.nis,
        kelas: verifiedSiswa.kelas,
        tanggalTugas: tanggalTugas,
        mapel: mapelInput,
        namaFile: finalFileName,
        waktuPengiriman: waktuKirim + ' WIB',
        status: 'Terkirim'
      }
    };
  } catch (err) {
    try { lock.releaseLock(); } catch (e) {}
    return { status: false, message: 'Gagal memproses pengumpulan tugas. Silakan coba kembali.' };
  }
}

// 5. Histori Pengiriman Tugas Khusus Siswa yang Login
function getHistoriTugasSiswa(token) {

// 6. Cek Status Pengumpulan Berdasarkan NIS dan ID Pengumpulan (Tanpa Mengungkap Data Siswa Lain)
function cekStatusTugas(nis, idPengumpulan) {
  const cleanNis = String(nis || '').trim().toLowerCase();
  const cleanId = String(idPengumpulan || '').trim().toUpperCase();

  if (!cleanNis || !cleanId || cleanId.length < 8) {
    return { status: false, message: 'Masukkan NIS dan Nomor ID Pengumpulan yang valid.' };
  }

  if (!checkRateLimit('CEK_TUGAS_' + cleanNis, 10)) {
    return { status: false, message: 'Terlalu banyak permintaan pengecekan. Coba lagi nanti.' };
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('PENGUMPULAN_TUGAS');
  if (!sheet) return { status: false, message: 'Data pengumpulan tidak ditemukan.' };

  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    const rId = String(data[i][0] || '').trim().toUpperCase();
    const rNis = String(data[i][2] || '').trim().toLowerCase();

    if (rId === cleanId && rNis === cleanNis) {
      return {
        status: true,
        receipt: {
          idPengumpulan: String(data[i][0] || ''),
          waktuPengiriman: data[i][1] instanceof Date ? Utilities.formatDate(data[i][1], TZ_WIB, 'yyyy-MM-dd HH:mm:ss') + ' WIB' : String(data[i][1] || ''),
          nis: String(data[i][2] || ''),
          namaSiswa: String(data[i][3] || ''),
          kelas: String(data[i][4] || ''),
          tanggalTugas: data[i][5] instanceof Date ? Utilities.formatDate(data[i][5], TZ_WIB, 'yyyy-MM-dd') : String(data[i][5] || '').substring(0, 10),
          mapel: String(data[i][6] || ''),
          namaFile: String(data[i][8] || ''),
          status: String(data[i][10] || 'Terkirim')
        }
      };
    }
  }

  return { status: false, message: 'Data tidak ditemukan. Pastikan NIS dan Nomor ID Pengumpulan sesuai.' };
}

// === PRESENSI GPS TERINTEGRASI KALENDER PENDIDIKAN & PEMISAHAN SHEET ===
function hitungJarakMeter(lat1, lon1, lat2, lon2) {
  const R = 6371000;
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
            Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
            Math.sin(dLon / 2) * Math.sin(dLon / 2);
  return R * (2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a)));
}

function simpanKehadiranGPS(token, payload) {
  try {
    const sess = verifySessionToken(token);
    if (!sess) return { status: false, message: 'Sesi login telah berakhir. Silakan masuk ulang.' };

    const now = new Date();
    const serverToday = Utilities.formatDate(now, TZ_WIB, 'yyyy-MM-dd');
    const waktuFormatted = Utilities.formatDate(now, TZ_WIB, 'HH:mm');

    // 1. Periksa Status Kalender Pendidikan Hari Ini
    const kalender = getKalenderStatusByDate(serverToday);
    if (kalender.statusEfektif.toLowerCase() === 'tidak efektif' || kalender.modeKbm.toLowerCase() === 'libur') {
      return { status: false, message: `Presensi ditutup. Hari ini terjadwal ${kalender.modeKbm} (${kalender.statusEfektif}).` };
    }

    // 2. Periksa Rentang Jam Efektif
    if (kalender.jamMulai && kalender.jamSelesai && kalender.jamMulai !== '-' && kalender.jamSelesai !== '-') {
      if (waktuFormatted < kalender.jamMulai || waktuFormatted > kalender.jamSelesai) {
        return {
          status: false,
          message: `Di luar jam efektif KBM (${kalender.jamMulai} - ${kalender.jamSelesai} WIB). Waktu server saat ini: ${waktuFormatted} WIB.`
        };
      }
    }

    // 3. Validasi Radius GPS berdasarkan Mode Luring vs Daring
    let infoJarak = 'Daring (GPS Nonaktif)';
    if (kalender.modeKbm.toLowerCase() === 'luring') {
      if (!payload.userLat || !payload.userLng) {
        return { status: false, message: 'KBM Luring mewajibkan deteksi lokasi GPS.' };
      }
      const cfg = getPengaturanSekolah();
      const targetLat = cfg ? cfg.latGps : -7.001759;
      const targetLng = cfg ? cfg.lngGps : 107.623047;
      const maxRadius = cfg ? cfg.radiusMeter : 100;

      const jarakRounded = Math.round(hitungJarakMeter(payload.userLat, payload.userLng, targetLat, targetLng));
      if (jarakRounded > maxRadius) {
        return { status: false, message: `Presensi Luring Gagal! Jarak Anda ${jarakRounded} meter (Maksimal ${maxRadius} meter).` };
      }
      infoJarak = `${jarakRounded} meter (Luring)`;
    } else {
      // Mode Daring: Radius GPS tidak membatasi presensi
      if (payload.userLat && payload.userLng) {
        const cfg = getPengaturanSekolah();
        const targetLat = cfg ? cfg.latGps : -7.001759;
        const targetLng = cfg ? cfg.lngGps : 107.623047;
        const jarakDaring = Math.round(hitungJarakMeter(payload.userLat, payload.userLng, targetLat, targetLng));
        infoJarak = `Daring (${jarakDaring}m - Bebas Radius)`;
      }
    }

    pemeriksaanSheetOtomatis();
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // 4. Pisahkan penyimpanan antara SISWA dan PEGAWAI (Admin/Guru/Karyawan)
    if (sess.role === 'siswa') {
      const sheetSiswaHadir = ss.getSheetByName('ABSENSI_DATA_KEHADIRAN');
      const recSiswa = {
        tglAbsen: serverToday,
        nis: sess.nis,
        nama: sess.nama,
        kelas: sess.kelas,
        jurusan: sess.jurusan || '-',
        jk: sess.jk || '-',
        status: 'Hadir',
        jarakMeter: infoJarak
      };
      const res = simpanAtauUpdateAbsenSiswa(sheetSiswaHadir, recSiswa);
      return {
        status: true,
        updated: res.updated,
        jarakMeter: infoJarak,
        waktuInput: waktuFormatted,
        tglAbsen: serverToday,
        message: `Presensi Siswa (${kalender.modeKbm}) Berhasil pada ${serverToday} pukul ${waktuFormatted} WIB!`
      };
    } else {
      const sheetPegawaiHadir = ss.getSheetByName('ABSENSI_KEHADIRAN_PEGAWAI');
      const recPeg = {
        tglAbsen: serverToday,
        nipUser: sess.nis || sess.username,
        nama: sess.nama,
        roleJabatan: sess.role.toUpperCase(),
        jk: sess.jk || 'L',
        status: 'Hadir',
        jarakMeter: infoJarak
      };
      const res = simpanAtauUpdateAbsenPegawai(sheetPegawaiHadir, recPeg);
      return {
        status: true,
        updated: res.updated,
        jarakMeter: infoJarak,
        waktuInput: waktuFormatted,
        tglAbsen: serverToday,
        message: `Presensi Pegawai (${kalender.modeKbm}) Berhasil pada ${serverToday} pukul ${waktuFormatted} WIB!`
      };
    }
  } catch (err) {
    return { status: false, message: 'Gagal menyimpan presensi. Silakan coba lagi.' };
  }
}

function simpanAtauUpdateAbsenSiswa(sheet, record) {
  const data = sheet.getDataRange().getValues();
  const now = new Date();
  const waktuJam = Utilities.formatDate(now, TZ_WIB, 'HH:mm');
  const serverToday = Utilities.formatDate(now, TZ_WIB, 'yyyy-MM-dd');
  const targetTgl = String(record.tglAbsen || serverToday).trim().substring(0, 10);
  const targetNis = String(record.nis || '').trim();

  let foundRow = -1;
  for (let i = 1; i < data.length; i++) {
    const rowTgl = data[i][0] instanceof Date ? Utilities.formatDate(data[i][0], TZ_WIB, 'yyyy-MM-dd') : String(data[i][0] || '').trim().substring(0, 10);
    const rowNis = String(data[i][2] || '').trim();
    if (rowTgl === targetTgl && rowNis.toLowerCase() === targetNis.toLowerCase()) {
      foundRow = i + 1;
      break;
    }
  }

  if (foundRow !== -1) {
    sheet.getRange(foundRow, 2).setValue(waktuJam);
    sheet.getRange(foundRow, 5).setValue(record.kelas || '');
    sheet.getRange(foundRow, 8).setValue(record.status || 'Hadir');
    if (record.jarakMeter !== undefined) sheet.getRange(foundRow, 9).setValue(record.jarakMeter);
    return { updated: true, waktuInput: waktuJam, tglAbsen: targetTgl };
  } else {
    sheet.appendRow([targetTgl, waktuJam, targetNis, record.nama, record.kelas || '', record.jurusan || '', record.jk || '-', record.status || 'Hadir', record.jarakMeter !== undefined ? record.jarakMeter : '-']);
    return { updated: false, waktuInput: waktuJam, tglAbsen: targetTgl };
  }
}
